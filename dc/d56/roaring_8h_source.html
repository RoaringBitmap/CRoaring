1/*
2 * An implementation of Roaring Bitmaps in C.
3 *
4 * This is the main public header for the 32-bit CRoaring API. A Roaring bitmap
5 * represents a set of unsigned 32-bit integers by partitioning the value space
6 * into 16-bit chunks and storing each chunk in a container chosen to match the
7 * local data density. Sparse chunks are typically kept as sorted arrays,
8 * denser chunks as bitsets, and long consecutive runs as run containers.
9 *
10 * This hybrid representation aims to keep bitmaps compact while still
11 * supporting fast membership tests, iteration, rank/select queries,
12 * serialization, and set operations such as union, intersection, difference,
13 * and symmetric difference.
14 */

102/*
103 * Whether you want to use copy-on-write.
104 * Saves memory and avoids copies, but needs more care in a threaded context.
105 * Most users should ignore this flag.
106 *
107 * Note: If you do turn this flag to 'true', enabling COW, then ensure that you
108 * do so for all of your bitmaps, since interactions between bitmaps with and
109 * without COW is unsafe.
110 *
111 * When setting this flag to false, if any containers are shared, they
112 * are unshared (cloned) immediately.
113 */
114inline bool roaring_bitmap_get_copy_on_write(const roaring_bitmap_t *r) {
115 return r->high_low_container.flags & ROARING_FLAG_COW;
116}
117inline void roaring_bitmap_set_copy_on_write(roaring_bitmap_t *r, bool cow) {
118 if (cow) {
119 r->high_low_container.flags |= ROARING_FLAG_COW;
120 } else {
121 if (roaring_bitmap_get_copy_on_write(r)) {
122 roaring_unshare_all(r);
123 }
124 r->high_low_container.flags &= ~ROARING_FLAG_COW;
125 }
126}

487inline bool roaring_bitmap_contains(const roaring_bitmap_t *r, uint32_t val) {
488 // For performance reasons, this function is inline and uses internal
489 // functions directly.
490#ifdef __cplusplus
491 using namespace ::roaring::internal;
492#endif
493 const uint16_t hb = val >> 16;
494 /*
495 * the next function call involves a binary search and lots of branching.
496 */
497 int32_t i = ra_get_index(&r->high_low_container, hb);
498 if (i < 0) return false;
499
500 uint8_t typecode;
501 // next call ought to be cheap
502 container_t *container = ra_get_container_at_index(&r->high_low_container,
503 (uint16_t)i, &typecode);
504 // rest might be a tad expensive, possibly involving another round of binary
505 // search
506 return container_contains(container, val & 0xFFFF, typecode);
507}

848/*
849 * "Frozen" serialization format imitates memory layout of roaring_bitmap_t.
850 * Deserialized bitmap is a constant view of the underlying buffer.
851 * This significantly reduces amount of allocations and copying required during
852 * deserialization.
853 * It can be used with memory mapped files.
854 * Example can be found in benchmarks/frozen_benchmark.c
855 *
856 * [#####] const roaring_bitmap_t *
857 * | | |
858 * +----+ | +-+
859 * | | |
860 * [#####################################] underlying buffer
861 *
862 * Note that because frozen serialization format imitates C memory layout
