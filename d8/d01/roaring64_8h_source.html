1/*
2 * roaring64.h
3 *
4 * This file declares the 64-bit Roaring bitmap API. A roaring64 bitmap stores
5 * sets of 64-bit unsigned integers by partitioning the value space by high
6 * bits and using Roaring containers for the lower bits inside each partition.
7 * This keeps the structure compact while preserving fast membership tests,
8 * insertions, iteration, and set operations over large sparse integer sets.
9 */

107// Use an immediately invoked closure, capturing by reference
108// (in case __VA_ARGS__ refers to context outside the closure)
109// Include a 0 at the beginning of the array to make the array length > 0
110// (zero sized arrays are not valid in standard c/c++)
111#define roaring64_bitmap_from(...) \
112 [&]() { \
113 const uint64_t roaring64_bitmap_from_array[] = {0, __VA_ARGS__}; \
114 return roaring64_bitmap_of_ptr( \
115 (sizeof(roaring64_bitmap_from_array) / \
116 sizeof(roaring64_bitmap_from_array[0])) - \
117 1, \
118 &roaring64_bitmap_from_array[1]); \
119 }()
120#else
127// While __VA_ARGS__ occurs twice in expansion, one of the times is in a sizeof
128// expression, which is an unevaluated context, so it's even safe in the case
129// where expressions passed have side effects (roaring64_bitmap_from(my_func(),
130// ++i))
131// Include a 0 at the beginning of the array to make the array length > 0
132// (zero sized arrays are not valid in standard c/c++)
133#define roaring64_bitmap_from(...) \
134 roaring64_bitmap_of_ptr( \
135 (sizeof((const uint64_t[]){0, __VA_ARGS__}) / sizeof(uint64_t)) - 1, \
136 &((const uint64_t[]){0, __VA_ARGS__})[1])
137#endif
